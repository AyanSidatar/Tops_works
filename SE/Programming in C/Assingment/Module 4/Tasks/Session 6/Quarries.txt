    user_id, 
    COUNT(order_id) AS order_count
FROM 
    orders
GROUP BY 
    user_id;
	
2.--SELECT 
    restaurant_name
FROM 
    restaurants
GROUP BY 
    restaurant_name
HAVING 
    AVG(rating) > 4.0;

    user_id

    payments

    user_id IN (
        SELECT 
            user_id 
        FROM 
            payments 
        WHERE 
            amount > 2000
    );
