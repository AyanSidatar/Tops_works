1.--CREATE TABLE influencers (
    influencer_id INT PRIMARY KEY,name VARCHAR(50)
);

	CREATE TABLE posts (
		post_id INT PRIMARY KEY,
		influencer_id INT,
		caption VARCHAR(255),
		FOREIGN KEY (influencer_id)
			REFERENCES influencers(influencer_id)
);

2.--SELECT 
    p.caption, 
    i.name AS influencer_name
FROM 
    posts p
INNER JOIN 
    influencers i ON p.influencer_id = i.influencer_id;
	
3.--SELECT 
    i.name AS influencer_name, 
    COALESCE(p.caption, 'No Posts') AS caption
FROM 
    influencers i
LEFT JOIN 
    posts p ON i.influencer_id = p.influencer_id;
	
4.--SELECT 
    p.caption, 
    i.name AS influencer_name
FROM 
    influencers i
RIGHT JOIN 
    posts p ON i.influencer_id = p.influencer_id;
